# INSTRUCCIONES DE INSTALACIÓN RÁPIDA
# ====================================

# 1. Verificar versiones instaladas:
# java -version  # Debe ser 17+
# node -v        # Debe ser 18+
# npm -v         # Debe ser 9+
# mvn -version   # Debe ser 3.6+

# 2. Instalar dependencias del backend:
# cd backend
# mvn clean install

# 3. Instalar dependencias del frontend:
# cd frontend
# npm install

# 4. Instalar Angular CLI globalmente (opcional):
# npm install -g @angular/cli@17

# NOTAS ADICIONALES
# =================

# - Todas las dependencias del backend se gestionan automáticamente por Maven
# - Las dependencias del frontend se instalan con npm install
# - El proyecto incluye configuración CORS para desarrollo local
# - La base de datos requiere configuración previa (ver database/script.sql)
# - Puerto del backend: 8082 (configurable en application.properties)
# - Puerto del frontend: 4200 (desarrollo) / configurable para producción
